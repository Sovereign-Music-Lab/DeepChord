+/// Single-use, process-local custody for the explicitly prepared physical target.
+/// This qualifies an exact raw-row edit only; it does not mint canonical timing
+/// or generic hidden-region/traversal proofs for the upstream MIDI reader.
+enum ObservedPitchEdit {
+    struct Failure: Error { let reason: String; init(_ reason: String) { self.reason = reason } }
+    struct Prepared: @unchecked Sendable {
+        let target: RegionEventPreparation.Target
+        let window: AXUIElement
+        let region: AXUIElement
+        let baseline: SelectedRegionEventListReader.PartialRead
+        let expires: Date
+    }
+    private static let lock = NSLock()
+    nonisolated(unsafe) private static var prepared: [String: Prepared] = [:]
+
+    static func retain(target: RegionEventPreparation.Target, window: AXUIElement,
+                       region: AXUIElement, baseline: SelectedRegionEventListReader.PartialRead) -> String? {
+        guard baseline.rawRows.count <= 10000 else { return nil }
+        lock.lock(); defer { lock.unlock() }
+        prepared = prepared.filter { $0.value.expires > Date() }
+        if prepared.count >= 8, let oldest = prepared.min(by: { $0.value.expires < $1.value.expires })?.key {
+            prepared.removeValue(forKey: oldest)
+        }
+        let id = UUID().uuidString
+        prepared[id] = Prepared(target: target, window: window, region: region, baseline: baseline,
+                                expires: Date().addingTimeInterval(300))
+        return id
+    }
+    static func consume(_ id: String) -> Prepared? {
+        lock.lock(); defer { lock.unlock() }
+        guard let value = prepared.removeValue(forKey: id), value.expires > Date() else { return nil }
+        return value
+    }

+    /// Logic's AX pitch setter can advance by one semitone, not jump to the
+    /// requested number. Every next write requires an observed monotonic step.
+    static func step(from initial: Int, to target: Int, read: () throws -> Int,
+                     write: (Int) throws -> Void, validate: () throws -> Void) throws -> Int {
+        guard (0...127).contains(initial), (0...127).contains(target), initial != target else {
+            throw Failure("invalid_pitch_change")
+        }
+        var current = initial
+        var writes = 0
+        while current != target {
+            try validate()
+            guard try read() == current else { throw Failure("pitch_changed_before_write") }
+            try write(target); writes += 1
+            try validate()
+            let next = try read()
+            let expectedStep = current + (target > current ? 1 : -1)
+            guard next == target || next == expectedStep else { throw Failure("unexpected_pitch_write_outcome") }
+            current = next
+            guard writes <= abs(target - initial) else { throw Failure("pitch_step_limit") }
+        }
+        try validate()
+        guard try read() == target else { throw Failure("final_pitch_changed") }
+        return writes
+    }
+
+    static func execute(params: [String: Value], runtime: AXLogicProElements.Runtime = .production) -> String {
+        var wroteNote = false
+        do {
+            guard Set(params.keys) == ["preparation_id", "row_index", "expected_pitch", "pitch"],
+                  case .string(let id) = params["preparation_id"],
+                  case .int(let index) = params["row_index"], index >= 0,

+    static func advance(from initial: Int, to target: Int, range: ClosedRange<Int>, limit: Int = 512,
+                        read: () throws -> Int, write: (Int) throws -> Void,
+                        validate: () throws -> Void) throws -> Int {
+        guard range.contains(initial), range.contains(target), limit > 0 else { throw Failure("invalid_position_step") }
+        var current = initial
+        var writes = 0
+        while current != target {
+            try validate()
+            guard try read() == current, writes < limit else { throw Failure("position_changed_or_step_limit") }
+            try write(target); writes += 1
+            try validate()
+            let next = try read()
+            guard next == target || next == current + (target > current ? 1 : -1) else {
+                throw Failure("unexpected_position_step")
+            }
+            current = next
+        }
+        try validate()
+        guard try read() == target else { throw Failure("position_changed_after_seek") }
+        return writes
+    }

+    static func advanceRebinding<Control>(to target: Int, range: ClosedRange<Int>,
+                                         resolve: () throws -> Control, read: (Control) throws -> Int,
+                                         write: (Control, Int) throws -> Void,
+                                         validate: () throws -> Void) throws -> Int {
+        let initial = try read(resolve())
+        var observed = initial
+        return try advance(from: initial, to: target, range: range, read: {
+            observed = try read(resolve())
+            return observed
+        }, write: { requested in
+            let fresh = try resolve()
+            guard try read(fresh) == observed else { throw Failure("position_changed_before_rebound_write") }
+            try write(fresh, requested)
+        }, validate: validate)
+    }
